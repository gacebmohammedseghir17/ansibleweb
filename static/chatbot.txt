<script>
    // ================= CONFIGURATION =================
    const CONFIG = {
        API_BASE_URL: 'http://169.254.147.45:1234/v1',  // LM Studio API endpoint
        MODEL_NAME: 'deepseek-r1-distill-llama-8b',
        SYSTEM_PROMPT: 'You are an expert Ansible assistant. Provide clear, concise answers about Ansible automation, playbooks, modules, and best practices. Always format code examples properly with syntax highlighting.',
        REQUEST_TIMEOUT: 30000,
        MAX_RETRIES: 3,
        BASE_DELAY: 1000,
        CORS_HEADERS: {
            'Content-Type': 'application/json',
            'Access-Control-Allow-Origin': '*',
            'Access-Control-Allow-Methods': 'GET, POST, OPTIONS',
            'Access-Control-Allow-Headers': 'Content-Type'
        }
    };

    // ================= DOM ELEMENTS =================
    const chatMessages = document.getElementById('chatMessages');
    const userInput = document.getElementById('userInput');
    const sendButton = document.getElementById('sendButton');
    const connectionStatus = document.getElementById('connectionStatus');
    const statusText = document.getElementById('statusText');
    const retryButton = document.getElementById('retryButton');
    const buttonText = document.getElementById('buttonText');
    const buttonIcon = document.getElementById('buttonIcon');

    // ================= STATE MANAGEMENT =================
    let messageHistory = [
        { role: 'assistant', content: "Hello! I'm your Ansible AI assistant. How can I help with your automation tasks today?" }
    ];
    let isProcessing = false;
    let isConnected = false;

    // ================= UTILITY FUNCTIONS =================
    function formatMessageContent(content) {
        // Enhanced markdown formatting
        return content
            .replace(/```(\w*)([\s\S]*?)```/g, '<pre><code>$2</code></pre>')
            .replace(/`([^`]+)`/g, '<code>$1</code>')
            .replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>')
            .replace(/\*(.*?)\*/g, '<em>$1</em>')
            .replace(/^### (.*$)/gm, '<h3>$1</h3>')
            .replace(/^## (.*$)/gm, '<h2>$1</h2>')
            .replace(/^# (.*$)/gm, '<h1>$1</h1>')
            .replace(/\n/g, '<br>');
    }

    function addMessage(content, isUser, isLoading = false) {
        const messageDiv = document.createElement('div');
        messageDiv.className = `message ${isUser ? 'user-message' : 'bot-message'}`;
        
        if (isLoading && !isUser) {
            messageDiv.innerHTML = `
                <div class="typing-indicator">
                    <div class="typing-dot"></div>
                    <div class="typing-dot"></div>
                    <div class="typing-dot"></div>
                </div>
            `;
        } else {
            messageDiv.innerHTML = formatMessageContent(content);
        }
        
        chatMessages.appendChild(messageDiv);
        chatMessages.scrollTop = chatMessages.scrollHeight;
        return messageDiv;
    }

    function autoResizeTextarea() {
        userInput.style.height = 'auto';
        userInput.style.height = (userInput.scrollHeight) + 'px';
    }

    function updateButtonState(isSending) {
        if (isSending) {
            sendButton.disabled = true;
            buttonText.textContent = 'Sending';
            buttonIcon.className = 'fas fa-spinner fa-spin';
        } else {
            sendButton.disabled = false;
            buttonText.textContent = 'Send';
            buttonIcon.className = 'fas fa-paper-plane';
        }
    }

    function updateConnectionStatus(connected, message = '') {
        isConnected = connected;
        if (connected) {
            connectionStatus.className = 'connection-status connected';
            statusText.innerHTML = `<i class="fas fa-check-circle"></i> Connected to LM Studio`;
            retryButton.style.display = 'none';
        } else {
            connectionStatus.className = 'connection-status disconnected';
            statusText.innerHTML = `<i class="fas fa-exclamation-circle"></i> ${message || 'Cannot connect to LM Studio'}`;
            retryButton.style.display = 'flex';
        }
    }

    // ================= CONNECTION HANDLING =================
    async function checkApiConnection() {
        updateConnectionStatus(false, 'Testing connection...');
        
        try {
            const response = await fetch(`${CONFIG.API_BASE_URL}/models`, {
                method: 'GET',
                headers: { 'Content-Type': 'application/json' },
                timeout: 5000
            });
            
            if (response.ok) {
                const data = await response.json();
                const modelAvailable = data.data?.some(model => model.id === CONFIG.MODEL_NAME);
                
                if (modelAvailable) {
                    updateConnectionStatus(true);
                    return true;
                } else {
                    updateConnectionStatus(false, `Model ${CONFIG.MODEL_NAME} not loaded in LM Studio`);
                    return false;
                }
            } else {
                throw new Error(`HTTP ${response.status}`);
            }
        } catch (error) {
            console.error("Connection check failed:", error);
            
            let errorMessage = 'Cannot connect to LM Studio';
            if (error.message.includes('Failed to fetch')) {
                errorMessage = 'Network error - check if LM Studio is running';
            } else if (error.message.includes('timeout')) {
                errorMessage = 'Connection timeout - check your network';
            }
            
            updateConnectionStatus(false, errorMessage);
            return false;
        }
    }

    // ================= MESSAGE HANDLING =================
    async function sendMessage() {
        const message = userInput.value.trim();
        if (!message || isProcessing) return;

        // Check connection if not already connected
        if (!isConnected) {
            const connected = await checkApiConnection();
            if (!connected) {
                addMessage("Cannot send message - please check your connection to LM Studio", false);
                return;
            }
        }

        isProcessing = true;
        updateButtonState(true);
        userInput.value = '';
        
        // Add user message to chat and history
        addMessage(message, true);
        messageHistory.push({ role: 'user', content: message });
        
        // Add loading indicator
        const loadingMessage = addMessage('', false, true);

        let retryCount = 0;
        let success = false;

        while (retryCount < CONFIG.MAX_RETRIES && !success) {
            try {
                const controller = new AbortController();
                const timeoutId = setTimeout(() => controller.abort(), CONFIG.REQUEST_TIMEOUT);

                const response = await fetch(`${CONFIG.API_BASE_URL}/chat/completions`, {
                    method: 'POST',
                    headers: CONFIG.CORS_HEADERS,
                    credentials: 'omit',
                    body: JSON.stringify({
                        messages: [
                            { role: 'system', content: CONFIG.SYSTEM_PROMPT },
                            ...messageHistory
                        ],
                        model: CONFIG.MODEL_NAME,
                        temperature: 0.7,
                        max_tokens: 2000,
                        stream: false
                    }),
                    signal: controller.signal
                });

                clearTimeout(timeoutId);

                if (!response.ok) {
                    const errorData = await response.json().catch(() => ({}));
                    throw new Error(errorData.error?.message || `HTTP ${response.status}`);
                }

                const data = await response.json();
                if (data.choices?.[0]?.message) {
                    const botMessage = data.choices[0].message;
                    messageHistory.push(botMessage);
                    loadingMessage.remove();
                    addMessage(botMessage.content, false);
                    success = true;
                } else {
                    throw new Error('Invalid response format');
                }
            } catch (error) {
                console.error(`Attempt ${retryCount + 1} failed:`, error);
                retryCount++;

                if (retryCount === CONFIG.MAX_RETRIES) {
                    loadingMessage.remove();
                    addMessage(`
                        Failed to get response after ${CONFIG.MAX_RETRIES} attempts.<br>
                        <strong>Error:</strong> ${error.message}<br><br>
                        Please verify:
                        <ul>
                            <li>LM Studio is running with API enabled</li>
                            <li>Model <code>${CONFIG.MODEL_NAME}</code> is loaded</li>
                            <li>API URL is correct: <code>${CONFIG.API_BASE_URL}</code></li>
                        </ul>
                    `, false);
                } else {
                    const delay = CONFIG.BASE_DELAY * Math.pow(2, retryCount - 1);
                    await new Promise(resolve => setTimeout(resolve, delay));
                }
            }
        }

        isProcessing = false;
        updateButtonState(false);
    }

    // ================= EVENT LISTENERS =================
    userInput.addEventListener('keydown', function(e) {
        if (e.key === 'Enter' && !e.shiftKey) {
            e.preventDefault();
            sendMessage();
        }
    });

    userInput.addEventListener('input', function() {
        autoResizeTextarea();
    });

    sendButton.addEventListener('click', sendMessage);

    retryButton.addEventListener('click', async function() {
        await checkApiConnection();
    });

    // ================= INITIALIZATION =================
    document.addEventListener('DOMContentLoaded', async function() {
        // Initial connection check
        await checkApiConnection();
        
        // Focus the input field
        userInput.focus();
    });

    // Make functions available for debugging
    window.debug = {
        checkApiConnection,
        getConfig: () => CONFIG,
        getState: () => ({ isConnected, messageHistory })
    };
</script>
